2วิเคราะห์เชิงพื้นที่ (Spatial Analysis & Filtering)

เมื่อได้ข้อมูลจุดความร้อนระบบจะใช้ Turf.js สร้างพิกัดและตรวจสอบเชิงเรขาคณิตว่าจุดนั้นตกลงในรูปหลายเหลี่ยม (Polygon) ซึ่งเป็นขอบเขต 4 จังหวัด (นครราชสีมา ชัยภูมิ บุรีรัมย์ สุรินทร์) หรือไม่

let turfPt = turf.point([lon, lat]); // แปลงพิกัดเป็น Turf Point
let strictProvName = null;

if (boundaryGeoJSON && boundaryGeoJSON.features) {
    for (let b of boundaryGeoJSON.features) {
        // เช็คก่อนว่าพิกัดอยู่ในกรอบสี่เหลี่ยม (BBox) ของจังหวัดนั้นไหม เพื่อลดภาระการประมวลผล
        if (lon >= b.bbox[0] && lat >= b.bbox[1] && lon <= b.bbox[2] && lat <= b.bbox[3]) {
            // เช็คความแม่นยำระดับเส้นขอบเขตจังหวัดว่าพิกัดตกอยู่ข้างในจริงๆ หรือไม่
            if (turf.booleanPointInPolygon(turfPt, b)) {
                let propStr = JSON.stringify(b.properties);
                for (let target of targetProvs) {
                    if (propStr.includes(target)) {
                        strictProvName = target; // พบว่าอยู่ใน 4 จังหวัดเป้าหมาย
                        break;
                    }
                }
            }
        }
        if (strictProvName) break;
    }
}

3ประเมินระดับความเสี่ยง (Risk Assessment)

นำพิกัดสถานที่กำจัดขยะมูลฝอยมาคำนวณระยะทางเทียบกับจุดความร้อนโดยใช้โมดูล distance ของ Turf.js หากพบว่าอยู่ห่างกันไม่เกิน 0.5 กิโลเมตรจะถือเป็นพื้นที่เสี่ยงไฟไหม้บ่อขยะ

const RISK_RADIUS_KM = 0.5; // รัศมี 500 เมตร

function calculateWasteFireRisk() {
    let risksFound = [];
    
    // วนลูปบ่อขยะทุกแห่ง
    landfillMarkers.forEach(site => {
        let sitePt = turf.point([site.marker.getLatLng().lng, site.marker.getLatLng().lat]);
        let nearHotspots = 0;

        // วนลูปจุดความร้อนที่ผ่านการกรองแล้ว
        currentFilteredHotspots.forEach(f => {
            let hotPt = turf.point([f.geometry.coordinates[0], f.geometry.coordinates[1]]);
            // คำนวณระยะห่างระหว่างบ่อขยะกับจุดความร้อนเป็นกิโลเมตร
            let distance = turf.distance(sitePt, hotPt, {units: 'kilometers'});
            
            // ถ้าระยะห่างอยู่ในเกณฑ์ความเสี่ยงให้นับเพิ่ม
            if (distance <= RISK_RADIUS_KM) {
                nearHotspots++;
            }
        });

        if (nearHotspots > 0) {
            risksFound.push({
                siteName: site.name,
                latlng: site.marker.getLatLng(),
                count: nearHotspots
            });
        }
    });
    displayFireRiskAlerts(risksFound);
}

4วิเคราะห์การแพร่กระจาย (Plume Dispersion & Downwind Impact)

ดึงข้อมูลพารามิเตอร์ด้านลมจาก Open-Meteo API มาสร้างรูปร่างควันไฟ (Plume) ไปยังพื้นที่ท้ายลมและแปลงพิกัดกลับเป็นที่อยู่โดยใช้ Longdo Map API เพื่อระบุชุมชนที่ได้รับผลกระทบ

// 1. ดึงข้อมูลสภาพอากาศและลม ณ พิกัดเกิดเหตุ
fetch(`https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current=wind_speed_10m,wind_direction_10m`)
    .then(res => res.json())
    .then(data => {
        let speed = data.current.wind_speed_10m;
        let dir = data.current.wind_direction_10m;
        
        // กลับทิศทางลมเพื่อหาทิศที่ควันจะลอยไป (พื้นที่ท้ายลม)
        const smokeAngle = (dir + 180) % 360;
        const bearing = smokeAngle > 180 ? smokeAngle - 360 : smokeAngle; 

        // 2. ใช้ Turf.js สร้างพิกัดเป้าหมายระยะ 2.5 กม และ 5 กม ตามทิศทางลม
        const pt1 = turf.destination([lon, lat], 2.5, bearing, {units: 'kilometers'}).geometry.coordinates;
        const pt2 = turf.destination([lon, lat], 5, bearing, {units: 'kilometers'}).geometry.coordinates;

        // 3. นำพิกัดปลายทางไปแปลงเป็นที่อยู่ด้วย Longdo API (Reverse Geocoding)
        Promise.all([
            fetch(`https://api.longdo.com/map/services/address?lon=${pt1[0]}&lat=${pt1[1]}&key=${longdoApiKey}`).then(r => r.json()),
            fetch(`https://api.longdo.com/map/services/address?lon=${pt2[0]}&lat=${pt2[1]}&key=${longdoApiKey}`).then(r => r.json())
        ]).then(results => {

5แจ้งเตือนและตอบสนอง (Alerting System)

สร้างองค์ประกอบส่วนต่อประสานกราฟิกเพื่อเร่งรัดกระบวนการตอบสนองซึ่งรวมถึงการวาดวงกลมแจ้งเตือนความเสี่ยงบนแผนที่และระบบ Toast Notification

function displayFireRiskAlerts(risks) {
    if (risks.length > 0) {
        risks.forEach(risk => {
            // วาดวงกลมสีแดงกระพริบทับพิกัดบ่อขยะเพื่อเน้นย้ำความสำคัญ
            L.circle(risk.latlng, {
                radius: RISK_RADIUS_KM * 1000, // แปลงรัศมีเป็นหน่วยเมตร
                color: '#d50000', 
                fillColor: '#ff1744', 
                fillOpacity: 0.3, 
                weight: 3,
                className: 'blink-alert' // เรียกใช้คลาส CSS ที่มีแอนิเมชันกระพริบ
            }).addTo(riskAlertLayer)
              .bindPopup(`<div style="text-align:center;"><b>🚨 พื้นที่เสี่ยงสูง!</b><br>บ่อขยะ: <span style="color:#b91c1c; font-weight:bold;">${risk.siteName}</span><br>พบจุดความร้อน ${risk.count} จุด ในรัศมี ${RISK_RADIUS_KM} กม.</div>`);
        });
        
        // แจ้งเตือนมุมขวาบนหน้าจอเพื่อให้เจ้าหน้าที่สังเกตเห็นทันที
        showToast(`⚠️ เฝ้าระวังฉุกเฉิน พบจุดความร้อนใกล้บ่อขยะ ${risks.length} แห่ง`, "error");
    }
}
